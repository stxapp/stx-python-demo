# Pip-installable pin for the demo scripts.
#
# Run:
#     pip install -r requirements.txt
#
# stx-python is in pre-release on TestPyPI until 1.0:
#   --index-url + --extra-index-url point pip at TestPyPI for the SDK
#     while still resolving transitive deps from main PyPI.
#   `>=0.0.0a0` is a pip idiom that allows pre-release matches for
#     this one package only. We *deliberately* don't use the global
#     `--pre` flag — that would cascade and let pip resolve transitive
#     deps (gql, graphql-core, …) to alpha/beta versions too, which
#     pulled a graphql-core 3.3 alpha that broke the SDK.
#
# Once we ship 1.0 to main PyPI, this whole file collapses to:
#     stx-python
--index-url https://test.pypi.org/simple/
--extra-index-url https://pypi.org/simple/

stx-python>=0.4.0a1
